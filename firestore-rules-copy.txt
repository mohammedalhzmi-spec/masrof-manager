rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function managerEmail() { return signedIn() && request.auth.token.email == 'alhzmim57@gmail.com'; }
    function hasProfile() { return signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)); }
    function profile() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    function isAdmin() { return managerEmail() || (hasProfile() && profile().active == true && profile().role == 'SYSTEM_ADMIN'); }
    function isFinance() { return isAdmin() || (hasProfile() && profile().active == true && profile().role == 'FINANCE_DIRECTOR'); }
    function isAccountant() { return hasProfile() && profile().active == true && profile().role == 'ACCOUNTANT'; }

    match /users/{userId} {
      allow get: if isAdmin() || (signedIn() && request.auth.uid == userId);
      allow list: if isAdmin();
      // إنشاء ملف المستخدم يتم مباشرة بعد Firebase Auth؛ لا نعتمد على role
      // في شرط الإنشاء حتى لا تُرفض التهيئة الأولى بسبب اختلاف تسمية الدور.
      allow create: if signedIn() && userId == request.auth.uid
        && request.resource.data.email == request.auth.token.email;
      allow update: if isAdmin() || (signedIn() && request.auth.uid == userId && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['fullName', 'updatedAt']));
      allow delete: if isAdmin() && userId != request.auth.uid;
    }

    // Contains only the public login alias and Firebase auth email; no passwords or financial data.
    match /authAliases/{aliasId} {
      allow read: if true;
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow update: if isAdmin() || (signedIn() && resource.data.uid == request.auth.uid);
      allow delete: if isAdmin();
    }

    match /devices/{deviceId} {
      allow read: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid);
      allow create: if signedIn() && request.resource.data.userId == request.auth.uid
        && request.resource.data.status in ['PENDING', 'APPROVED'];
      allow update: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['fcmToken', 'lastSeenAt']));
      allow delete: if isAdmin();
    }

    match /accessRequests/{requestId} {
      allow read: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid);
      allow create: if signedIn() && request.resource.data.userId == request.auth.uid && request.resource.data.status == 'PENDING';
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }

    match /documents/{docId} {
      allow read: if signedIn() && hasProfile() && profile().active == true;
      allow create: if signedIn() && hasProfile() && profile().active == true && request.resource.data.createdByUid == request.auth.uid;
      allow update: if isFinance() || (signedIn() && resource.data.createdByUid == request.auth.uid && resource.data.status in ['DRAFT', 'SUBMITTED']);
      allow delete: if false;
    }

    match /auditLogs/{logId} {
      allow read: if isAdmin();
      allow create: if signedIn() && request.resource.data.actorUid == request.auth.uid;
      allow update, delete: if false;
    }

    match /notifications/{notificationId} {
      allow read: if isAdmin() || (signedIn() && resource.data.userId == request.auth.uid);
      allow create: if isAdmin();
      allow update: if signedIn() && (isAdmin() || resource.data.userId == request.auth.uid);
      allow delete: if isAdmin();
    }
  }
}
